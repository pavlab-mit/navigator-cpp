add_library(nav_gpio OBJECT gpio.cpp)
target_include_directories(nav_gpio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(nav_gpio PRIVATE ${CMAKE_SOURCE_DIR}/include)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(CheckSymbolExists)
    set(CMAKE_REQUIRED_LIBRARIES gpiod)
    check_symbol_exists(gpiod_chip_request_lines "gpiod.h" NAV_HAVE_LIBGPIOD_V2)
    unset(CMAKE_REQUIRED_LIBRARIES)
    if(NAV_HAVE_LIBGPIOD_V2)
        target_compile_definitions(nav_gpio PRIVATE NAV_LIBGPIOD_V2=1)
        message(STATUS "navigator-cpp: libgpiod v2 API")
    else()
        message(STATUS "navigator-cpp: libgpiod v1 API")
    endif()
endif()
